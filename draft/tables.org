* Processing times
Note that this only takes the first and last parquet files for each directory. Also we may have used a different number of instances when running the models.
#+begin_src sh :eval noexport :exports results
  ls -lU --time-style=+%s /work/datasets/aliby_output/*/jump_core_annotated/jpegxl_lossy_mq.zarr/profiles |
    awk '
    BEGIN {printf "*Model* *sites/s* *Hours(total)*\n"};
    /^\// {
      group_starts=NR;
      if (NR>1) names[s[5]]=max-min;
      split($0, s, "/")
    } 
    {if (NR == group_starts+2) min = max = $6}
    /parquet/ {
    if ($6 < min) min = $6;
    if ($6 > max) max = $6;
    }
    END {
  names[s[5]]=max-min;
  for (key in names) printf("%s %s %s\n", key, 726720/names[key], names[key]/3600)}
    ' 
#+end_src
